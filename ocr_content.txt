import json

transcript_path = r"C:\Users\HP\.gemini\antigravity-ide\brain\759e5eea-54f9-4681-918e-c8791f81a713\.system_generated\logs\transcript_full.jsonl"

with open(transcript_path, "r", encoding="utf-8") as f:
    for i, line in enumerate(f):
        if "==Start of OCR" in line:
            print(f"Line {i} matches!")
            data = json.loads(line)
            # Recursively print key paths containing ==Start of OCR
            def search_val(v, path=""):
                if isinstance(v, dict):
                    for k, val in v.items():
                        search_val(val, f"{path}.{k}")
                elif isinstance(v, list):
                    for idx, val in enumerate(v):
                        search_val(val, f"{path}[{idx}]")
                elif isinstance(v, str):
                    if "==Start of OCR" in v:
                        print(f"Path: {path}, len: {len(v)}")
                        # write to ocr_content.txt
                        with open("ocr_content.txt", "w", encoding="utf-8") as out:
                            out.write(v)
            search_val(data)
