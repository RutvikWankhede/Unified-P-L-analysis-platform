import json, re

transcript_path = r"C:\Users\HP\.gemini\antigravity-ide\brain\759e5eea-54f9-4681-918e-c8791f81a713\.system_generated\logs\transcript_full.jsonl"
with open(transcript_path, "r", encoding="utf-8") as f:
    for i, line in enumerate(f):
        if "TXN-00001" in line:
            obj = json.loads(line)
            tool_calls = obj.get("tool_calls", [])
            for tc in tool_calls:
                args = tc.get("args", {})
                code = args.get("CodeContent", "")
                if "TXN-00001" in code:
                    print(f"Line {i} has code of length {len(code)}")
                    with open("extracted_ocr.txt", "w", encoding="utf-8") as out:
                        out.write(code)
                    print("Saved to extracted_ocr.txt")
